   for(id in ids){val t=p.unit(4)!!.sessions.flatMap{it.steps}.first{id in it.itemIds};val i=p.item(id)!!;assertNull(i.assessment);s.v2RecordGovernedAttempt(prepareGovernedAttempt(p,id,i.answer,PRESCRIBED_CHECK,t.id,"2026-10-14T00:00")!!)}
   assertEquals(ids.toSet(),creditItemIds(p,s.v2LearnerView()));assertEquals(V2ObjectiveState.DEMONSTRATED_IN_APP_CHECKS,objectiveState(o,s.v2LearnerView(),p));assertEquals(3,s.v2Attempts().size)
  }
 }
 @Test fun namedRoleObjectiveMustHaveDistinctIndependentCaseRoleEvidence() {

 }

}
