package dev.russiancompanion.v2
import android.content.Context
import androidx.test.core.app.ApplicationProvider
import dev.russiancompanion.data.local.LocalStore
import dev.russiancompanion.v2.curriculum.*
import org.junit.Assert.*
import org.junit.Test
import org.junit.runner.RunWith
import org.robolectric.RobolectricTestRunner
import org.robolectric.annotation.Config
import java.io.File
@RunWith(RobolectricTestRunner::class)
@Config(sdk=[33])
class Unit13ReportedBaselineTest {
 @Test fun originalSixAnswersCompleteBothObjectivesWithEarlierIndependentThenLaterGuide() {
  val p=pack();val u=p.unit(13)!!
  assertTrue(u.sessions.indexOfFirst{it.id=="u13-s-tense-keep"}<u.sessions.indexOfFirst{it.id=="u13-s-reported"})
  LocalStore.openInMemory(ApplicationProvider.getApplicationContext<Context>()).use { store ->
   for(id in listOf("obj-u13-tense-keep","obj-u13-reported")) {
    val o=u.objectives.single{it.id==id};val ids=o.practiceStepIds.flatMap{p.step(it)!!.second.itemIds}+o.checkItemIds;assertEquals(3,ids.size)
    for(item in ids) {val step=u.sessions.flatMap{it.steps}.single{item in it.itemIds};store.v2RecordGovernedAttempt(prepareGovernedAttempt(p,item,p.item(item)!!.answer,PRESCRIBED_CHECK,step.id,"2026-10-13T23:00")!!)}
    assertEquals(V2ObjectiveState.DEMONSTRATED_IN_APP_CHECKS,objectiveState(o,store.v2LearnerView(),p))
   }
   assertEquals(6,store.v2Attempts().size)
  }
 }
 @Test fun laterGuideNeedsDifferentContextFromTheEarlierEveningCheck() {
  val p=pack();assertNotEquals("IA016: later guide repeats earlier independent context",p.item("u13-i-tense-keep-1")!!.promptLetter,p.item("u13-g-reported-0")!!.promptLetter)
 }
 @Test fun reportedObjectiveNeedsItsOwnIndependentFrameAndEmbeddedTimeInterpretation() {
  val p=pack();val o=p.unit(13)!!.objectives.single{it.id=="obj-u13-reported"}
  assertTrue("Selecting direct/Moscow labels alone cannot establish reported-clause tense interpretation",o.checkItemIds.any{p.item(it)!!.assessment?.testedFeature=="REPORTED_FRAME_EMBEDDED_TIME"})
 }
 private fun pack()=V2PackLoader.parse(File(if(File("app").isDirectory)"." else "..","app/src/main/assets/content/v2-curriculum.json").readText())
}
