package dev.russiancompanion.v2
import android.content.Context
import androidx.test.core.app.ApplicationProvider
import dev.russiancompanion.data.local.LocalStore
import dev.russiancompanion.v2.curriculum.*
import dev.russiancompanion.v2.backup.*
import org.junit.Assert.*
import org.junit.Test
import org.junit.runner.RunWith
import org.robolectric.RobolectricTestRunner
import org.robolectric.annotation.Config
import kotlinx.coroutines.runBlocking
import java.io.File
@RunWith(RobolectricTestRunner::class)
@Config(sdk=[33])
class CumulativeSelectionBaselineExportTest {
 @Test fun exportGenuineSchemaOneCompletedHistoryBeforeSelectorChange() {
  val p=V2PackLoader.parse(File(root(),"app/src/main/assets/content/v2-curriculum.json").readText())
  V2PackIdentityIndex.warm(p)
  ready(p).use { s ->
   val a=s.openCumulativeReview(p,3,s.learnerMutationEpoch())!!
   assertEquals(1,a.schema)
   assertEquals("u1-i-р-not-p",a.slots[1].itemId)
   a.slots.forEachIndexed { n,slot -> assertTrue(s.answerCumulativeReview(p,a.id,n,p.item(slot.itemId)!!.answer,s.learnerMutationEpoch())) }
   runBlocking {
    val session=p.unit(4)!!.sessions.first().id
    assertTrue(s.v2AdvanceUnitIfCurrentAsync(s.learnerMutationEpoch(),3,"baseline-completion",4,session,resumePositionDigest(p,session,0)!!,GOVERNED_IDENTITY_SCHEMA,p))
   }
   assertEquals(setOf(1,2,3),currentCompletedUnitNumbers(p,s.v2LearnerView()))
   File(root(),"audit/independent-gpt61-2026-10-12/ia003-recent-selection/BASELINE_SCHEMA1_HISTORY.json").writeText(backupJson.encodeToString(GovernedLearnerDocument.serializer(),s.governedLearnerDocument()))
  }
 }
    private fun ready(p:V2CurriculumPack)=LocalStore.openInMemory(context()).also { s ->
        for(n in 1..3) {
            s.v2SaveProfile(s.v2Profile().copy(onboardingCompleted=true,currentUnitNumber=n,currentSessionId=null))
            for(step in p.unit(n)!!.sessions.flatMap { it.steps }) {
                step.itemIds.forEach { id -> s.v2RecordGovernedAttempt(prepareGovernedAttempt(p,id,p.item(id)!!.answer,PRESCRIBED_CHECK,step.id,"same-clock")!!) }
                s.v2CompleteGovernedStep(step.id,governedStepCompletionDigest(p,s.v2LearnerView(),step.id)!!,"same-clock")
            }
            s.v2AttestBook(n,"same-clock"); if(n<3) s.v2CompleteUnit(n,"same-clock")
        }
    }

 private fun context():Context=ApplicationProvider.getApplicationContext()
 private fun root()=File(if(File("app").isDirectory)"." else "..")
}
