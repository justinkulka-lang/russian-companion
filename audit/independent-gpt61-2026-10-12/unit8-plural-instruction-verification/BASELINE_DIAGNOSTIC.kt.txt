package dev.russiancompanion.v2
import dev.russiancompanion.v2.curriculum.*
import org.junit.Assert.*
import org.junit.Test
import java.io.File
class Unit8PluralBaselineTest {
 @Test fun everyPluralTeachingCardMustPrecedePracticeAndReturnInRemediation() {
  val p=V2PackLoader.parse(File(if(File("app").isDirectory)"." else "..","app/src/main/assets/content/v2-curriculum.json").readText());val s=p.unit(8)!!.sessions.single{it.id=="u8-s-plural"};val t=s.steps.filter{it.kind=="TEACH"}
  assertEquals("IA011: seven plural teaching cards occur after guided practice",t.size,s.steps.takeWhile{it.kind!="GUIDED"}.count{it.kind=="TEACH"})
  assertEquals(t.joinToString("\n\n"){it.body},s.steps.single{it.kind=="REMEDIATE"}.body)
 }
}
