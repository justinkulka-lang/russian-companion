import java.nio.file.*;
import java.util.*;
public class GenderPackValidator {
  public static void main(String[] args) throws Exception {
    Object pack = ReviewCheckpointTools.method("dev.russiancompanion.v2.curriculum.V2PackLoader", "parse", 1)
      .invoke(ReviewCheckpointTools.singleton("dev.russiancompanion.v2.curriculum.V2PackLoader"),
        Files.readString(Path.of("/workspace/russian-companion/app/src/main/assets/content/v2-curriculum.json")));
    Object result = ReviewCheckpointTools.method("dev.russiancompanion.v2.curriculum.V2Validator", "validate", 2)
      .invoke(ReviewCheckpointTools.singleton("dev.russiancompanion.v2.curriculum.V2Validator"), pack, null);
    List<?> errors = (List<?>) result.getClass().getMethod("getErrors").invoke(result);
    if (!errors.isEmpty()) throw new IllegalStateException(errors.toString());
    System.out.println("Pack validation PASS; zero errors; no JUnit methods executed.");
  }
}
